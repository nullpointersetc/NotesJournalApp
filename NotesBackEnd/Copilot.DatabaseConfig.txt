public class NotesBackEnd
{
    private static void Main(string[] args)
    {
        WebApplicationBuilder builder =
            WebApplication.CreateBuilder(args);

        // Load all connections from appsettings.json
        var allConnections =
            builder.Configuration.GetSection("Connections")
                .Get<Dictionary<string, ConnectionConfig>>();

        if (allConnections is null || allConnections.Count == 0)
        {
            Console.WriteLine("No connections found in configuration.");
            return;
        }

        // Parse command-line argument: --db=Name
        string? requested = args
            .FirstOrDefault(a => a.StartsWith("--db=", StringComparison.OrdinalIgnoreCase))
            ?.Substring("--db=".Length);

        // Select connection or fall back to "default"
        ConnectionConfig selected =
            requested != null && allConnections.TryGetValue(requested, out var cfg)
                ? cfg
                : allConnections["default"];

        Console.WriteLine($"Using connection: {requested ?? "default"}");
        Console.WriteLine($"DatabaseType: {selected.DatabaseType}");

        // Register DbContext based on DatabaseType
        if (selected.DatabaseType.Equals("SqlServer", StringComparison.OrdinalIgnoreCase))
        {
            builder.Services.AddDbContext<NotesDbContextForSqlServer>(
                options => options.UseSqlServer(selected.ConnectionString));
        }
        else if (selected.DatabaseType.Equals("Sqlite", StringComparison.OrdinalIgnoreCase))
        {
            builder.Services.AddDbContext<NotesDbContextForSqlite>(
                options => options.UseSqlite(selected.ConnectionString));
        }
        else
        {
            throw new InvalidOperationException(
                $"Unknown DatabaseType: {selected.DatabaseType}");
        }

        // Register handlers + repositories
        builder.Services.AddScoped<INoteHandler, NoteHandler>();
        builder.Services.AddScoped<IUserHandler, UserHandler>();
        builder.Services.AddScoped<INoteRepository, NoteRepository>();
        builder.Services.AddScoped<IUserRepository, UserRepository>();

        builder.Services.AddEndpointsApiExplorer();
        builder.Services.AddSwaggerGen();

        WebApplication app = builder.Build();

        NotesRestAPI.MapEndpoints(app);
        UsersRestAPI.MapEndpoints(app);

        if (app.Environment.IsDevelopment())
        {
            app.UseSwagger();
            app.UseSwaggerUI();
        }

        app.Run();
    }
}
