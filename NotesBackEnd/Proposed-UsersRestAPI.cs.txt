using Microsoft.AspNetCore.Builder;
using Microsoft.AspNetCore.Http;
using NullPointersEtc.NotesJournalApp.NotesHandlers;
using NullPointersEtc.NotesJournalApp.UserEntity;
using System;

namespace NullPointersEtc.NotesJournalApp.NotesBackEnd;

public static class UsersRestAPI
{
    public static void MapEndpoints(WebApplication app)
    {
        // ---------------------------------------------------------
        // CREATE USER
        // ---------------------------------------------------------
        app.MapPost("/api/users", async (
            IUserHandler handler,
            CreateUserDTO dto) =>
        {
            User user = await handler.CreateUserWithHandlerAsync(
                dto.UserName,
                dto.DisplayName,
                dto.EMailAddress);

            return Results.Created($"/api/users/{user.UserID}",
                new UserDTO(
                    user.UserID,
                    user.UserName,
                    user.DisplayName,
                    user.EMailAddress,
                    user.CreatedAt,
                    user.LastModifiedAt));
        });

        // ---------------------------------------------------------
        // GET USER BY ID
        // ---------------------------------------------------------
        app.MapGet("/api/users/{userID:guid}", async (
            IUserHandler handler,
            Guid userID) =>
        {
            User user = await handler.GetUserFromUserIdWithHandlerAsync(userID);

            return Results.Ok(new UserDTO(
                user.UserID,
                user.UserName,
                user.DisplayName,
                user.EMailAddress,
                user.CreatedAt,
                user.LastModifiedAt));
        });

        // ---------------------------------------------------------
        // GET USER BY USERNAME
        // ---------------------------------------------------------
        app.MapGet("/api/users/by-username/{userName}", async (
            IUserHandler handler,
            string userName) =>
        {
            User user = await handler.GetUserFromUserNameWithHandlerAsync(userName);

            return Results.Ok(new UserDTO(
                user.UserID,
                user.UserName,
                user.DisplayName,
                user.EMailAddress,
                user.CreatedAt,
                user.LastModifiedAt));
        });

        // ---------------------------------------------------------
        // GET USER BY DISPLAY NAME
        // ---------------------------------------------------------
        app.MapGet("/api/users/by-displayname/{displayName}", async (
            IUserHandler handler,
            string displayName) =>
        {
            User user = await handler.GetUserFromDisplayNameWithHandlerAsync(displayName);

            return Results.Ok(new UserDTO(
                user.UserID,
                user.UserName,
                user.DisplayName,
                user.EMailAddress,
                user.CreatedAt,
                user.LastModifiedAt));
        });

        // ---------------------------------------------------------
        // GET ALL USERS
        // ---------------------------------------------------------
        app.MapGet("/api/users", async (IUserHandler handler) =>
        {
            var users = await handler.GetAllUsersWithHandlerAsync();

            var dtos = users.Select(u =>
                new UserDTO(
                    u.UserID,
                    u.UserName,
                    u.DisplayName,
                    u.EMailAddress,
                    u.CreatedAt,
                    u.LastModifiedAt));

            return Results.Ok(dtos);
        });

        // ---------------------------------------------------------
        // UPDATE USER
        // ---------------------------------------------------------
        app.MapPut("/api/users/{userID:guid}", async (
            IUserHandler handler,
            Guid userID,
            UpdateUserDTO dto) =>
        {
            User updated = await handler.UpdateUserWithHandlerAsync(
                userID,
                dto.DisplayName,
                dto.EMailAddress);

            return Results.Ok(new UserDTO(
                updated.UserID,
                updated.UserName,
                updated.DisplayName,
                updated.EMailAddress,
                updated.CreatedAt,
                updated.LastModifiedAt));
        });

        // ---------------------------------------------------------
        // DELETE USER
        // ---------------------------------------------------------
        app.MapDelete("/api/users/{userID:guid}", async (
            IUserHandler handler,
            Guid userID) =>
        {
            await handler.DeleteWithHandlerAsync(userID);
            return Results.NoContent();
        });
    }
}

// ---------------------------------------------------------
// DTOs
// ---------------------------------------------------------

public sealed record CreateUserDTO(
    string UserName,
    string DisplayName,
    string EMailAddress);

public sealed record UpdateUserDTO(
    string DisplayName,
    string EMailAddress);

public sealed record UserDTO(
    Guid UserID,
    string UserName,
    string DisplayName,
    string EMailAddress,
    DateTime CreatedAt,
    DateTime LastModifiedAt);
