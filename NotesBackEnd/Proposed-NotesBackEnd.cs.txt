using Microsoft.AspNetCore.Builder;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Configuration;
using Microsoft.EntityFrameworkCore;
using NullPointersEtc.NotesJournalApp.NotesStorage;
using NullPointersEtc.NotesJournalApp.NotesHandlers;
using NullPointersEtc.NotesJournalApp.NotesBackEnd;

public sealed class ConnectionConfig
{
    public string DatabaseType { get; set; } = "";
    public string ConnectionString { get; set; } = "";
}

public sealed class ConnectionsOptions
{
    public Dictionary<string, ConnectionConfig> Connections { get; set; }
        = new(StringComparer.OrdinalIgnoreCase);
}

public class NotesBackEnd
{
    private static void Main(string[] args)
    {
        WebApplicationBuilder builder = WebApplication.CreateBuilder(args);

        // Load "Connections" section into Options pattern
        builder.Services.Configure<ConnectionsOptions>(
            builder.Configuration.GetSection("Connections"));

        // Determine which connection to use
        string? requested =
            args.FirstOrDefault(a => a.StartsWith("--db=", StringComparison.OrdinalIgnoreCase))
                ?.Substring("--db=".Length);

        var allConnections =
            builder.Configuration.GetSection("Connections")
                .Get<Dictionary<string, ConnectionConfig>>();

        string envName = builder.Environment.EnvironmentName;

        string? envDefault =
            builder.Configuration[$"Connections:{envName}:DefaultConnection"];

        string selectedKey =
            requested
            ?? envDefault
            ?? "default";

        if (!allConnections.TryGetValue(selectedKey, out ConnectionConfig? selected))
        {
            selected = allConnections["default"];
            selectedKey = "default";
        }

        Console.WriteLine($"Environment: {envName}");
        Console.WriteLine($"Using connection: {selectedKey}");
        Console.WriteLine($"DatabaseType: {selected.DatabaseType}");

        // Register DbContext based on provider
        if (selected.DatabaseType.Equals("SqlServer", StringComparison.OrdinalIgnoreCase))
        {
            builder.Services.AddDbContext<NotesDbContext>(options =>
                options.UseSqlServer(selected.ConnectionString),
                contextLifetime: ServiceLifetime.Scoped);
        }
        else
        {
            builder.Services.AddDbContext<NotesDbContext>(options =>
                options.UseSqlite(selected.ConnectionString),
                contextLifetime: ServiceLifetime.Scoped);
        }

        // Register handlers + repositories
        builder.Services.AddScoped<INoteHandler, NoteHandler>();
        builder.Services.AddScoped<IUserHandler, UserHandler>();
        builder.Services.AddScoped<INoteRepository, NoteRepository>();
        builder.Services.AddScoped<IUserRepository, UserRepository>();

        builder.Services.AddEndpointsApiExplorer();
        builder.Services.AddSwaggerGen();

        WebApplication app = builder.Build();

        // Map endpoints
        NotesRestAPI.MapEndpoints(app);
        UsersRestAPI.MapEndpoints(app);

        // Automatic table creation / migration
        using (var scope = app.Services.CreateScope())
        {
            var db = scope.ServiceProvider.GetRequiredService<NotesDbContext>();

            if (db.Database.IsSqlite())
            {
                db.Database.EnsureCreated();
            }
            else
            {
                db.Database.Migrate();
            }
        }

        if (app.Environment.IsDevelopment())
        {
            app.UseSwagger();
            app.UseSwaggerUI();
        }

        app.Run();
    }
}
